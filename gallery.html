var SHEET_NAME = "GalleryData"; function setupSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } var headers = sheet.getRange(1, 1, 1, 3).getValues()[0]; if (headers[0] !== "Date" || headers[1] !== "Base64 Image Data" || headers[2] !== "File Name") { sheet.getRange(1, 1, 1, 3).setValues([["Date", "Base64 Image Data", "File Name"]]); sheet.getRange(1, 1, 1, 3).setFontWeight("bold"); sheet.setFrozenRows(1); } return sheet; } function doPost(e) { try { var params = JSON.parse(e.postData.contents); var sheet = setupSheet(); var dateStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss"); // Save Base64 directly into the sheet (No Drive used) sheet.appendRow([dateStr, params.base64, params.name]); return ContentService.createTextOutput(JSON.stringify({"success": true})).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({"success": false, "error": error.toString()})).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { var sheet = setupSheet(); var data = sheet.getDataRange().getValues(); var images = []; // Fetch Base64 data directly from the sheet for(var i = 1; i < data.length; i++) { if(data[i][1]) { images.push(data[i][1]); } } images.reverse(); return ContentService.createTextOutput(JSON.stringify(images)).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON); } }
